-- Art of War - old save compatibility migration
-- ===============================================

-- Purpose
-- -------

-- This migration upgrades a save created with the old five-table schema so the
-- current name-aware save loader can read it.

-- Assumptions
-- -----------

-- The input file is an old save with the tables units, buildings, resources,
-- players, and config.
-- Units are idle and are not moving.
-- Units have no target, pending target, aim, path, order, formation, or pending
-- action.
-- Buildings have no combat target.
-- No production queues, upgrades, projectiles, formations, pending commands,
-- AI history, camera state, or random-generator continuation are migrated.
-- Run this script once against a copy of the old save.

-- The ALTER TABLE statements append the missing columns after the old columns.
-- The current loader reads required save columns by name, so their physical
-- SQLite column order is not significant; appending keeps the migration simple.

-- Usage with sqlite3.exe
-- ----------------------

-- sqlite3.exe old-save.db < old-save-migration.txt

-- Recommended backup command:

-- copy old-save.db old-save.db.bak

-- Migration
-- ---------

-- Old units had 10 columns. Append the 20 new unit continuation columns.
ALTER TABLE units ADD COLUMN next_state INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN state_pending INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN frame_state INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN formation INT NOT NULL DEFAULT -1;
ALTER TABLE units ADD COLUMN pos_in_state INT NOT NULL DEFAULT -1;
ALTER TABLE units ADD COLUMN command_priority INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN command_center INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN charge_energy INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN target_uid INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_target_uid INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN aim_kind INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN aim_target_uid INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN aim_current INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN aim_direction_x INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN aim_direction_z INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_aim_kind INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_aim_target_uid INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_aim_current INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_aim_direction_x INT NOT NULL DEFAULT 0;
ALTER TABLE units ADD COLUMN pending_aim_direction_z INT NOT NULL DEFAULT 0;

-- Keep the old current state as the continuation state. state_pending remains 0,
-- so the loader will not apply a movement, target, or pending state transition.
UPDATE units SET next_state = state;

-- Old buildings had 10 columns. Append combat continuation fields. Deployment
-- cells are regenerated from the loaded map and are not appended.
ALTER TABLE buildings ADD COLUMN target_uid INT NOT NULL DEFAULT 0;
ALTER TABLE buildings ADD COLUMN frame_state INT NOT NULL DEFAULT 0;

-- Old players had 12 columns. Append mutable resource statistics.
ALTER TABLE players ADD COLUMN gather_food INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN gather_wood INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN gather_stone INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN gather_gold INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN pending_gather_food INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN pending_gather_wood INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN pending_gather_stone INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN pending_gather_gold INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN sum_food INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN sum_wood INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN sum_stone INT NOT NULL DEFAULT 0;
ALTER TABLE players ADD COLUMN sum_gold INT NOT NULL DEFAULT 0;

-- Old config had 3 columns. Start simulation time at tick zero.
ALTER TABLE config ADD COLUMN total_ticks INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_present INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_seed INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_ai_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_resource_rotation_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_collision_force_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_other_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_save_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_player_name_idx INT NOT NULL DEFAULT 0;
ALTER TABLE config ADD COLUMN rdn_resource_node_idx INT NOT NULL DEFAULT 0;

COMMIT;

-- Validation
-- ----------

-- Run these commands after the migration:

PRAGMA table_info(units);
PRAGMA table_info(buildings);
PRAGMA table_info(players);
PRAGMA table_info(config);

-- The loader requires these current column counts. The migrated database can
-- have more physical columns because named-column reads ignore legacy fields
-- that were removed from the current contract.

-- units: 30
-- buildings: 11
-- players: 22
-- config: 13

-- With the documented legacy input, the physical counts after this script are:
-- units: 30
-- buildings: 12 (the legacy deploy_Idx column is retained and ignored)
-- players: 24 (the legacy name and color columns are retained and ignored)
-- config: 13

-- The optional current-format runtime tables are intentionally not created by
-- this migration. The loader treats them as absent and starts their state
-- empty. RNG continuation is represented by config in the current schema;
-- camera state is intentionally recreated on load. AI continuation state remains
-- empty because it is stored in optional
-- ai_state, ai_wants, and ai_history tables.
